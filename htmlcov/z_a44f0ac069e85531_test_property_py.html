1"""Property-based tests for input-facing parsers. 

2 

3The other repo's "OSS-Fuzz has no added value" comment was directionally right 

4for pure-Python thin-wrapper libraries -- most of the crash class it would find 

5is cheaper to find with in-process property tests. This module runs 

6:mod:`hypothesis` on the parsers that accept the noisiest user input: 

7 

8- ``InvoiceTemplate.parse_number`` -- locale-dependent number parsing 

9- ``InvoiceTemplate.parse_date`` -- format + language matrix via ``_dates`` 

10- ``ordered_load`` -- YAML/JSON template streams from DB/API payloads 

11- ``read_templates`` -- YAML files loaded from disk 

12- ``regex.compile`` on captured template patterns -- ReDoS guard 

13 

14Contract under test: the parsers may return sensible failure values or raise 

15one of the *typed* :class:`~invoice2data.exceptions.InvoiceProcessingError` 

16subclasses (which subclass :class:`ValueError`). They must NOT raise 

17``AssertionError``, ``TypeError`` (except the documented internal one), 

18``AttributeError``, ``UnicodeError``, or hang the process. 

19""" 

20

46def _tpl(**overrides: Any) -> InvoiceTemplate: 

47 """Build a minimal valid template so `parse_number` / `parse_date` work.""" 

48 base: dict[str, Any] = { 

49 "template_name": "hypothesis.yml", 

50 "keywords": ["Anything"], 

51 "exclude_keywords": [], 

52 "options": { 

53 "currency": "EUR", 

54 "decimal_separator": ".", 

55 "date_formats": [], 

56 "languages": ["en"], 

57 "replace": [], 

58 }, 

59 } 

60 base.update(overrides) 

61 return InvoiceTemplate(base)

65# parse_number # 

66# --------------------------------------------------------------------------- # 

67 

68_number_bytes = st.text( 

69 alphabet=st.sampled_from("0123456789.,'\t\n -+eE"), min_size=1, max_size=32 

70) 

71 

72 

73@given(value=_number_bytes) 

74@settings(max_examples=400, suppress_health_check=[HealthCheck.function_scoped_fixture]) 

75def test_parse_number_never_crashes_unexpectedly(value: str) -> None: 

76 """`parse_number` may raise ValueError (or subclass) or return a float. 

77 

78 It must NOT propagate an unexpected error class -- that's an internal 

79 invariant leak. 

80 """ 

81 tpl = _tpl() 

82 try: 

83 result = tpl.parse_number(value) 

84 except (ValueError, InvoiceProcessingError): 

85 return # documented contract 

86 else: 

87 assert isinstance(result, float), ( 

88 f"expected float, got {type(result).__name__} for input {value!r}" 

89 ) 

90 

91 

92@given(sep=st.sampled_from([".", ","])) 

93def test_parse_number_respects_decimal_separator(sep: str) -> None: 

94 """A number with the declared separator round-trips to a float.""" 

95 tpl = _tpl(options={**_tpl().options, "decimal_separator": sep}) 

96 assert tpl.parse_number(f"1{sep}25") == pytest.approx(1.25)

104@given(value=st.text(min_size=0, max_size=64)) 

105@settings(max_examples=200, suppress_health_check=[HealthCheck.function_scoped_fixture]) 

106def test_parse_date_never_crashes(value: str) -> None: 

107 """`parse_date` must return a datetime, a date, or None -- never raise.""" 

108 tpl = _tpl() 

109 try: 

110 result = tpl.parse_date(value) 

111 except InvoiceProcessingError: 

112 return # typed error class is allowed 

113 # None (no parse) or a datetime/date-like are the only allowed happy paths. 

114 assert result is None or hasattr(result, "year"), ( 

115 f"expected date-like or None, got {type(result).__name__}" 

116 )

153@settings(max_examples=100, suppress_health_check=[HealthCheck.function_scoped_fixture]) 

154def test_ordered_load_accepts_well_formed_stream( 

155 keywords: list[str], extra: dict[str, str] 

156) -> None: 

157 """A stream with the required `keywords` field always parses into a template.""" 

158 tpl_dict = {"issuer": "H", "keywords": keywords, **extra} 

159 assume(all(k != "template_name" for k in extra)) 

160 payload = json.dumps([tpl_dict]) 

161 

162 result = ordered_load(payload) 

163 assert len(result) == 1 

164 assert result[0]["keywords"] == keywords
