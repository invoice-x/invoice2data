1"""`alternatives:` on the lines parser (#759). 

2 

3The lines parser already had ``rules:`` for **concatenating** results from 

4multiple line-parsing sub-blocks (all rules run, results appended). This 

5adds a parallel ``alternatives:`` key with **first-non-empty-wins** 

6semantics, so template authors can express "try layout A; if A yields 

7nothing, fall back to layout B" instead of stuffing two blocks into one 

8mapping and losing the first to YAML key-collision. 

9""" 

10 

11import logging 

12 

13import pytest 

14 

15from invoice2data.extract.invoice_template import InvoiceTemplate 

16from invoice2data.extract.parsers import lines as lines_parser

19_TPL = InvoiceTemplate( 

20 [ 

21 ("issuer", "test"), 

22 ("keywords", ["test"]), 

23 ("exclude_keywords", []), 

24 ("template_name", "alternatives.yml"), 

25 ( 

26 "options", 

27 { 

28 "currency": "EUR", 

29 "languages": [], 

30 "decimal_separator": ".", 

31 "remove_whitespace": False, 

32 "remove_accents": False,

42_ALT_A = { 

43 "start": r"BEGIN A", 

44 "end": r"END A",

56_CONTENT_A_AND_B = "\n".join(["BEGIN A", "SKU1 1", "END A", "BEGIN B", "C2 2", "END B"])

60def test_first_alternative_matches_wins() -> None: 

61 """When both alternatives could match, the first that yields rows wins. 

62 

63 B is never consulted; results are exactly A's. 

64 """ 

65 settings = {"line_separator": r"\n", "alternatives": [_ALT_A, _ALT_B]} 

66 rows = lines_parser.parse(_TPL, "lines", settings, _CONTENT_A_AND_B) 

67 assert rows == [{"sku": "SKU1", "qty": "1"}]

70def test_first_alternative_empty_falls_back_to_second() -> None: 

71 """A's markers absent -> A yields nothing -> B runs and returns its rows.""" 

72 settings = {"line_separator": r"\n", "alternatives": [_ALT_A, _ALT_B]} 

73 rows = lines_parser.parse(_TPL, "lines", settings, _CONTENT_B_ONLY) 

74 assert rows == [{"code": "C99", "count": "5"}]

95def test_per_alternative_override_beats_shared_top_level() -> None: 

96 """An alternative's own key wins over the shared top-level value.""" 

97 # Top-level `line` would match SKU rows; alternative B overrides it to 

98 # match a completely different shape (Cxx rows), so on B-only content the 

99 # B alternative wins with its own `line` pattern -- not the shared one. 

100 settings = { 

101 "line_separator": r"\n", 

102 "line": r"^(?P<sku>SKU\d+)\s+(?P<qty>\d+)$", # shared default 

103 "alternatives": [ 

104 {"start": r"BEGIN A", "end": r"END A"}, # inherits shared `line` 

105 { 

106 "start": r"BEGIN B", 

107 "end": r"END B", 

108 "line": r"^(?P<code>C\d+)\s+(?P<count>\d+)$", # override 

109 }, 

110 ], 

111 } 

112 rows = lines_parser.parse(_TPL, "lines", settings, _CONTENT_B_ONLY) 

113 assert rows == [{"code": "C99", "count": "5"}]

116def test_alternatives_and_rules_together_warns_and_prefers_alternatives( 

117 caplog: pytest.LogCaptureFixture, 

118) -> None: 

119 """`alternatives:` wins over `rules:` at the same level; log a warning.""" 

120 settings = { 

121 "line_separator": r"\n", 

122 "rules": [ 

123 # If `rules` were honoured here it would concatenate and return 

124 # BOTH the A rows AND the B rows. 

125 _ALT_A, 

126 _ALT_B, 

127 ], 

128 "alternatives": [_ALT_A, _ALT_B], 

129 } 

130 with caplog.at_level(logging.WARNING, logger=lines_parser.__name__): 

131 rows = lines_parser.parse(_TPL, "lines", settings, _CONTENT_A_AND_B) 

132 # First-non-empty-wins -> only A's row, not A+B concatenated. 

133 assert rows == [{"sku": "SKU1", "qty": "1"}] 

134 assert any( 

135 "'alternatives' takes precedence" in rec.getMessage() for rec in caplog.records 

136 )
